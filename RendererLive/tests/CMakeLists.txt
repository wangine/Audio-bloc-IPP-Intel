cmake_minimum_required(VERSION 3.24)
project(EntropyLiveContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/RendererLive/Wdl.cmake")
add_executable(EntropyLiveContracts Contracts.cpp)
target_include_directories(EntropyLiveContracts PRIVATE "${ROOT}")
target_link_libraries(EntropyLiveContracts PRIVATE entropy_live_wdl)
if(WIN32)
 # The original PCM implementation targets Windows/MSVC (uint64 typedefs).
 # Do not edit its algorithm to satisfy an unrelated Linux portability build.
 add_library(entropy_contract_pcm STATIC "${ROOT}/Egress/PCM24/src/Pcm24Converter.cpp" "${ROOT}/Egress/PCM24/src/Pcm24ModuleEngine.cpp")
 target_include_directories(entropy_contract_pcm PUBLIC "${ROOT}/Shared/include" "${ROOT}/Egress/PCM24/include")
 target_sources(EntropyLiveContracts PRIVATE DomainContracts.cpp WorkersContracts.cpp
  "${ROOT}/Engine/Egress/LiveSinkWorkers.cpp" "${ROOT}/RendererLive/WasapiOutput.cpp")
 target_link_libraries(EntropyLiveContracts PRIVATE entropy_contract_pcm ole32 uuid avrt bcrypt)
 target_compile_definitions(EntropyLiveContracts PRIVATE ENTROPY_WINDOWS_DOMAIN_CONTRACTS=1)
 target_compile_definitions(entropy_contract_pcm PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(entropy_contract_pcm PRIVATE /W4 /EHsc /fp:precise /utf-8)
endif()
if(MSVC)
 target_compile_definitions(EntropyLiveContracts PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(EntropyLiveContracts PRIVATE /W4 /EHsc /fp:precise /utf-8)
else()
 target_compile_options(EntropyLiveContracts PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
