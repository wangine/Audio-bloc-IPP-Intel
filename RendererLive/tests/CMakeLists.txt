cmake_minimum_required(VERSION 3.24)
project(EntropyLiveContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/RendererLive/Wdl.cmake")
add_executable(EntropyLiveContracts Contracts.cpp)
target_include_directories(EntropyLiveContracts PRIVATE "${ROOT}")
target_link_libraries(EntropyLiveContracts PRIVATE entropy_live_wdl)
add_library(entropy_contract_pcm STATIC "${ROOT}/Egress/PCM24/src/Pcm24Converter.cpp" "${ROOT}/Egress/PCM24/src/Pcm24ModuleEngine.cpp")
target_include_directories(entropy_contract_pcm PUBLIC "${ROOT}/Shared/include" "${ROOT}/Egress/PCM24/include")
add_executable(EntropyDomainContracts DomainContracts.cpp)
target_include_directories(EntropyDomainContracts PRIVATE "${ROOT}")
target_link_libraries(EntropyDomainContracts PRIVATE entropy_live_wdl entropy_contract_pcm)
set(check_targets EntropyLiveContracts EntropyDomainContracts entropy_contract_pcm)
if(WIN32)
 add_executable(EntropyWorkersContracts WorkersContracts.cpp "${ROOT}/Engine/Egress/LiveSinkWorkers.cpp" "${ROOT}/RendererLive/WasapiOutput.cpp")
 target_include_directories(EntropyWorkersContracts PRIVATE "${ROOT}")
 target_link_libraries(EntropyWorkersContracts PRIVATE entropy_live_wdl entropy_contract_pcm ole32 uuid avrt bcrypt)
 list(APPEND check_targets EntropyWorkersContracts)
endif()
foreach(t ${check_targets})
 if(MSVC)
  target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${t} PRIVATE /W4 /EHsc /fp:precise /utf-8)
 else()
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
 endif()
endforeach()
