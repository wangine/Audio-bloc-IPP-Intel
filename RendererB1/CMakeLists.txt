cmake_minimum_required(VERSION 3.24)
project(EntropyRendererB1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "B1 Windows candidate requires MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Keep existing DSP/profile implementation intact. Old executables are not built.
set(BUILD_TESTING OFF CACHE BOOL "No general test suite in B1 build" FORCE)
set(ENTROPY_STAGE04_EGRESS ON CACHE BOOL "Existing profile dependencies" FORCE)
set(ENTROPY_STAGE05_CONTROL ON CACHE BOOL "Existing offline profile validation" FORCE)
set(ENTROPY_STAGE02_VALIDATION OFF CACHE BOOL "" FORCE)
set(ENTROPY_V07_INTEGRATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/existing" EXCLUDE_FROM_ALL)
add_executable(EntropyPrepareB1 Prepare.cpp)
target_link_libraries(EntropyPrepareB1 PRIVATE entropy_control_runtime entropy_pcm24)
add_executable(EntropyPlayB1 Play.cpp)
# This executable MUST NOT link the existing graph, egress, SRC, clock or Qt.
target_link_libraries(EntropyPlayB1 PRIVATE ole32 uuid avrt)
add_executable(EntropyPacketB1Tests Tests.cpp)
foreach(t EntropyPrepareB1 EntropyPlayB1 EntropyPacketB1Tests)
 target_compile_features(${t} PRIVATE cxx_std_20)
 target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
 target_compile_options(${t} PRIVATE /utf-8 /fp:precise /EHsc /W4)
endforeach()
