cmake_minimum_required(VERSION 3.22)
project(EntropyMusicRenderV2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME MusicRenderReference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/PackageTests.py --reference-only)
add_library(MusicRenderModel STATIC model/Format.cpp model/Clock.cpp model/PcmRing.cpp model/PacketTimeline.cpp protocol/PcmBridgeV1.cpp kernel/WaveRtTimeline.cpp)
target_compile_options(MusicRenderModel PRIVATE /W4 /WX /GR-)
foreach(suite Model Bridge KernelBinding)
    add_executable(MusicRender${suite}Tests tests/${suite}Tests.cpp)
    target_link_libraries(MusicRender${suite}Tests PRIVATE MusicRenderModel)
    target_compile_options(MusicRender${suite}Tests PRIVATE /W4 /WX)
    add_test(NAME MusicRender${suite} COMMAND MusicRender${suite}Tests)
endforeach()
add_executable(EntropyMusicPcmAudit audit/WasapiClient.cpp audit/main.cpp provider/WindowsEndpointIdentity.cpp)
target_link_libraries(EntropyMusicPcmAudit PRIVATE MusicRenderAuditCommon setupapi ole32 uuid)
target_compile_definitions(EntropyMusicPcmAudit PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(EntropyMusicPcmAudit PRIVATE /W4 /WX)
add_executable(MusicRenderAuditTests tests/AuditTests.cpp)
add_library(MusicRenderAuditCommon STATIC audit/PcmPattern.cpp audit/PcmVerifier.cpp audit/StereoMetrics.cpp)
target_link_libraries(MusicRenderAuditCommon PUBLIC MusicRenderProviderCommon)
target_compile_options(MusicRenderAuditCommon PRIVATE /W4 /WX)
target_link_libraries(MusicRenderAuditTests PRIVATE MusicRenderAuditCommon)
target_compile_options(MusicRenderAuditTests PRIVATE /W4 /WX)
add_test(NAME MusicRenderAudit COMMAND MusicRenderAuditTests)
add_library(MusicRenderProviderCommon STATIC provider/BridgeClient.cpp provider/EndpointIdentity.cpp)
target_link_libraries(MusicRenderProviderCommon PUBLIC MusicRenderModel)
target_compile_options(MusicRenderProviderCommon PRIVATE /W4 /WX)
add_executable(MusicRenderProviderTests tests/ProviderTests.cpp)
target_link_libraries(MusicRenderProviderTests PRIVATE MusicRenderProviderCommon)
target_compile_options(MusicRenderProviderTests PRIVATE /W4 /WX)
add_test(NAME MusicRenderProvider COMMAND MusicRenderProviderTests)
set_tests_properties(MusicRenderProvider PROPERTIES TIMEOUT 10)
add_library(EntropyMusicRawProviderV2 SHARED provider/RawMusicProvider.cpp provider/NativeBridge.cpp provider/WindowsEndpointIdentity.cpp provider/RawMusicProvider.def)
target_link_libraries(EntropyMusicRawProviderV2 PRIVATE MusicRenderProviderCommon setupapi ole32 uuid)
target_compile_definitions(EntropyMusicRawProviderV2 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(EntropyMusicRawProviderV2 PRIVATE /W4 /WX)
add_executable(MusicRenderObserverTests tests/ObserverTests.cpp)
target_link_libraries(MusicRenderObserverTests PRIVATE MusicRenderProviderCommon)
target_compile_options(MusicRenderObserverTests PRIVATE /W4 /WX)
add_test(NAME MusicRenderObserver COMMAND MusicRenderObserverTests)
add_library(EntropyMusicRawProviderV2Audit SHARED provider/RawMusicProvider.cpp provider/NativeBridge.cpp provider/WindowsEndpointIdentity.cpp provider/AuditObserver.cpp provider/RawMusicProvider.def)
target_link_libraries(EntropyMusicRawProviderV2Audit PRIVATE MusicRenderProviderCommon setupapi ole32 uuid)
target_compile_definitions(EntropyMusicRawProviderV2Audit PRIVATE ENTROPY_AUDIT_PROVIDER NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(EntropyMusicRawProviderV2Audit PRIVATE /W4 /WX)
add_executable(MusicRenderProviderLoadTests tests/ProviderLoadTests.cpp)
target_compile_definitions(MusicRenderProviderLoadTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(MusicRenderProviderLoadTests PRIVATE /W4 /WX)
foreach(provider EntropyMusicRawProviderV2 EntropyMusicRawProviderV2Audit)
    add_test(NAME ${provider}Load COMMAND MusicRenderProviderLoadTests $<TARGET_FILE:${provider}>)
endforeach()
