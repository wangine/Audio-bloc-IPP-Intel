cmake_minimum_required(VERSION 3.22)
project(EntropyMusicRenderV2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME MusicRenderReference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/PackageTests.py --reference-only)
add_library(MusicRenderModel STATIC model/Format.cpp model/Clock.cpp model/PcmRing.cpp model/PacketTimeline.cpp protocol/PcmBridgeV1.cpp)
target_compile_options(MusicRenderModel PRIVATE /W4 /WX /EHs-c- /GR-)
foreach(suite Model Bridge)
    add_executable(MusicRender${suite}Tests tests/${suite}Tests.cpp)
    target_link_libraries(MusicRender${suite}Tests PRIVATE MusicRenderModel)
    target_compile_options(MusicRender${suite}Tests PRIVATE /W4 /WX)
    add_test(NAME MusicRender${suite} COMMAND MusicRender${suite}Tests)
endforeach()
