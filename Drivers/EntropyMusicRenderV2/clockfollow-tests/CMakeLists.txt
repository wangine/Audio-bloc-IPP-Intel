cmake_minimum_required(VERSION 3.22)
project(EntropyClockFollowContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(ClockFollowModel STATIC ../model/Clock.cpp ../model/Format.cpp ../model/PcmRing.cpp ../model/PacketTimeline.cpp ../protocol/PcmBridgeV1.cpp ../kernel/WaveRtTimeline.cpp)
add_executable(EntropyClockFollowContracts ClockContracts.cpp LeaseContracts.cpp TimelineContracts.cpp)
target_compile_definitions(EntropyClockFollowContracts PRIVATE ENTROPY_CLOCKFOLLOW_LEASE_TESTS=1)
target_link_libraries(EntropyClockFollowContracts PRIVATE ClockFollowModel)
if(MSVC)
 target_compile_options(ClockFollowModel PRIVATE /W4 /WX /GR-)
 target_compile_options(EntropyClockFollowContracts PRIVATE /W4 /WX /EHsc)
 foreach(name Model Bridge KernelBinding Inbox)
  add_executable(ClockFollowLegacy${name} ../tests/${name}Tests.cpp)
  target_link_libraries(ClockFollowLegacy${name} PRIVATE ClockFollowModel)
  target_compile_options(ClockFollowLegacy${name} PRIVATE /W4 /WX /EHsc)
 endforeach()
else()
 target_compile_options(ClockFollowModel PRIVATE -Wall -Wextra -Werror -pedantic)
 target_compile_options(EntropyClockFollowContracts PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
