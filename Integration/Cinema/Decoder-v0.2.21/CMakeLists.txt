cmake_minimum_required(VERSION 3.24)
project(EntropySurroundDecoder VERSION 0.2.21 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/utf-8 /W4)
add_library(CinemaTransport STATIC ThirdParty/StagePublishV2/StagePublishV2.cpp ThirdParty/StageLink/StageLinkSecurity.cpp ThirdParty/StageBusSDK/src/StageBusValidation.cpp)
target_include_directories(CinemaTransport PUBLIC ThirdParty/StagePublishV2 ThirdParty/StageLink ThirdParty/StageBusSDK/include)
target_link_libraries(CinemaTransport PUBLIC advapi32 ole32)
add_executable(CinemaProbe tests/CinemaProbe.cpp)
target_link_libraries(CinemaProbe PRIVATE CinemaTransport)
add_library(EntropySurroundCore STATIC source/EntropySurroundCore.cpp)
target_include_directories(EntropySurroundCore PUBLIC source)
add_executable(EntropySurroundCoreTests source/EntropySurroundCoreTests.cpp)
target_link_libraries(EntropySurroundCoreTests PRIVATE EntropySurroundCore)
enable_testing()
add_test(NAME ExistingSurroundCore COMMAND EntropySurroundCoreTests)
add_library(CinemaOutput STATIC source/CinemaOutput.cpp source/CinemaWdlSrc.cpp ThirdParty/WDL/resample.cpp)
target_include_directories(CinemaOutput PUBLIC source ThirdParty/WDL)
target_compile_definitions(CinemaOutput PUBLIC WDL_RESAMPLE_FULL_SINC_PRECISION)
target_link_libraries(CinemaOutput PUBLIC CinemaTransport avrt)
add_executable(OutputTests tests/OutputTests.cpp)
target_link_libraries(OutputTests PRIVATE CinemaOutput)
add_test(NAME CinemaOutputNumerical COMMAND OutputTests)
add_executable(TransportTests tests/TransportTests.cpp)
target_link_libraries(TransportTests PRIVATE CinemaOutput)
add_test(NAME CinemaOutputTransport COMMAND TransportTests)
add_executable(SpectralTests tests/SpectralTests.cpp)
target_link_libraries(SpectralTests PRIVATE CinemaOutput)
add_test(NAME CinemaOutputSpectrum COMMAND SpectralTests)
add_executable(IsolationTests tests/IsolationTests.cpp)
target_link_libraries(IsolationTests PRIVATE CinemaOutput)
add_test(NAME CinemaOutputIsolation COMMAND IsolationTests)
add_executable(CapturePolicyTests tests/CapturePolicyTests.cpp)
target_link_libraries(CapturePolicyTests PRIVATE EntropySurroundCore)
add_test(NAME CinemaCapturePolicy COMMAND CapturePolicyTests)
add_executable(RetirementTests tests/RetirementTests.cpp)
target_link_libraries(RetirementTests PRIVATE CinemaOutput)
add_test(NAME CinemaProducerRetirement COMMAND RetirementTests)
add_executable(CadenceTests tests/CadenceTests.cpp)
target_link_libraries(CadenceTests PRIVATE CinemaOutput)
add_test(NAME CinemaPublicationCadence COMMAND CadenceTests)
add_executable(EntropySurroundDecoder source/EntropySurroundDecoder.cpp)
target_compile_definitions(EntropySurroundDecoder PRIVATE ENTROPY_SURROUND_VERSION="${PROJECT_VERSION}")
target_link_libraries(EntropySurroundDecoder PRIVATE EntropySurroundCore CinemaOutput ole32 uuid propsys avrt)
add_subdirectory("D:/Projects/JUCE" JUCE EXCLUDE_FROM_ALL)
juce_add_gui_app(EntropySurroundDecoderGUI PRODUCT_NAME "Entropy Surround Decoder" COMPANY_NAME "Entropy Audio" VERSION ${PROJECT_VERSION} BUNDLE_ID "audio.entropy.surrounddecoder" ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/entropy_icon.png" ICON_SMALL "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/entropy_icon.png")
juce_add_binary_data(EntropySurroundAssets SOURCES ThirdParty/entropy_icon.png)
target_sources(EntropySurroundDecoderGUI PRIVATE source/EntropySurroundGui.cpp)
target_compile_definitions(EntropySurroundDecoderGUI PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 ENTROPY_SURROUND_VERSION="${PROJECT_VERSION}")
target_link_libraries(EntropySurroundDecoderGUI PRIVATE EntropySurroundAssets juce::juce_gui_extra juce::juce_audio_basics)
