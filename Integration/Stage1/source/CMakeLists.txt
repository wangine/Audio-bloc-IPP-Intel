include(CTest)
get_filename_component(CMAKE_CXX_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
function(bundle_mingw t)
 if(NOT MINGW)
  return()
 endif()
 foreach(dll libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
  add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CXX_COMPILER_DIRECTORY}/${dll}" "$<TARGET_FILE_DIR:${t}>")
 endforeach()
endfunction()
