cmake_minimum_required(VERSION 3.22)
project(EntropySampleRateConverter LANGUAGES C CXX)

add_library(EntropySrcEngine STATIC src/MultiChannelSrcEngine.cpp)
target_include_directories(EntropySrcEngine PUBLIC include)
target_include_directories(EntropySrcEngine PRIVATE R8Brain/src/r8brain-free-src)
target_link_libraries(EntropySrcEngine PUBLIC EntropyModuleABI)
target_compile_features(EntropySrcEngine PUBLIC cxx_std_20)
target_compile_definitions(EntropySrcEngine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(EntropySrcEngine PRIVATE /utf-8 /W4 /fp:precise)
endif()

add_library(EntropySampleRateConverter SHARED src/SrcModule.cpp)
target_link_libraries(EntropySampleRateConverter PRIVATE EntropySrcEngine EntropyModuleABI)
target_compile_features(EntropySampleRateConverter PRIVATE cxx_std_20)
target_compile_definitions(EntropySampleRateConverter PRIVATE ENTROPY_MODULE_BUILD=1 NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(EntropySampleRateConverter PRIVATE /utf-8 /W4 /fp:precise)
endif()
set_target_properties(EntropySampleRateConverter PROPERTIES PREFIX "" OUTPUT_NAME "EntropySampleRateConverter")
option(ENTROPY_MODULE_COPY_TO_SOURCE "Copy built modules into source bin directories" ON)
if(ENTROPY_MODULE_COPY_TO_SOURCE)
add_custom_command(TARGET EntropySampleRateConverter POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_SOURCE_DIR}/bin/msvc-x64"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:EntropySampleRateConverter>
    "${CMAKE_CURRENT_SOURCE_DIR}/bin/msvc-x64/EntropySampleRateConverter.dll")
endif()

if(BUILD_TESTING)
  add_executable(EntropyNominalEgressContractTests tests/NominalEgressContractTests.cpp)
  target_link_libraries(EntropyNominalEgressContractTests PRIVATE EntropySrcEngine)
  add_test(NAME EntropyNominalEgressContract COMMAND EntropyNominalEgressContractTests)
  add_executable(EntropySrcModuleTests tests/SrcModuleTests.cpp)
  target_link_libraries(EntropySrcModuleTests PRIVATE EntropyModuleABI)
  target_compile_features(EntropySrcModuleTests PRIVATE cxx_std_20)
  target_compile_definitions(EntropySrcModuleTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_test(NAME EntropySrcModule COMMAND EntropySrcModuleTests $<TARGET_FILE:EntropySampleRateConverter>)

  add_executable(EntropySrcQualityTests tests/SrcQualityTests.cpp)
  target_link_libraries(EntropySrcQualityTests PRIVATE EntropySrcEngine)
  target_compile_features(EntropySrcQualityTests PRIVATE cxx_std_20)
  add_test(NAME EntropySrcQuality COMMAND EntropySrcQualityTests)

  add_executable(EntropySrcRealtimeAllocationTests tests/SrcRealtimeAllocationTests.cpp)
  target_link_libraries(EntropySrcRealtimeAllocationTests PRIVATE EntropySrcEngine)
  target_compile_features(EntropySrcRealtimeAllocationTests PRIVATE cxx_std_20)
  add_test(NAME EntropySrcRealtimeAllocations COMMAND EntropySrcRealtimeAllocationTests)
endif()
