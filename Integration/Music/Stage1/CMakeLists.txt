cmake_minimum_required(VERSION 3.22)
project(EntropyStage1Music LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
 add_compile_options(/utf-8 /W4 /fp:precise)
endif()
add_library(Stage1Pcm STATIC Engine/Ingress/PcmDecode.cpp)
target_include_directories(Stage1Pcm PUBLIC ABI Engine/Ingress)
add_library(Stage1NativeSrc STATIC ThirdParty/NativeSrc/Modules/SRC/src/MultiChannelSrcEngine.cpp)
target_include_directories(Stage1NativeSrc PUBLIC ThirdParty/NativeSrc/Modules/SRC/include ThirdParty/NativeSrc/Shared/ModuleABI/include)
add_executable(Stage1Acceptance tests/Stage1Acceptance.cpp)
add_library(Stage1Ingress STATIC Engine/Ingress/ProviderRegistry.cpp Engine/Ingress/WindowsEndpointCatalog.cpp)
target_include_directories(Stage1Ingress PUBLIC ABI Engine/Ingress)
target_link_libraries(Stage1Ingress PUBLIC ole32 uuid propsys)
add_library(Stage1Pipeline STATIC Engine/SRC/SourceDomain.cpp Engine/Matrix/InputMatrix.cpp Engine/Meters/SourcePeakMeter.cpp)
target_include_directories(Stage1Pipeline PUBLIC Engine/SRC Engine/Matrix Engine/Transport Engine/Meters)
target_link_libraries(Stage1Pipeline PUBLIC Stage1Pcm Stage1NativeSrc)
add_subdirectory(ThirdParty/StageTransport/StageBusSDK)
add_subdirectory(ThirdParty/StageTransport/StageLink)
add_subdirectory(ThirdParty/StageTransport/StagePublishV2)
add_library(Stage1Publisher STATIC Engine/Transport/MusicPublisher.cpp)
target_include_directories(Stage1Publisher PUBLIC Engine/Transport)
target_link_libraries(Stage1Publisher PUBLIC Stage1Pipeline EntropyStagePublishV2)
target_link_libraries(Stage1Acceptance PRIVATE Stage1Publisher Stage1Ingress)
target_compile_definitions(Stage1Acceptance PRIVATE TEST_PROVIDER_DIR="${CMAKE_BINARY_DIR}/test-providers")
target_compile_definitions(Stage1Acceptance PRIVATE TEST_ACTUAL_PROVIDER_DIR="${CMAKE_BINARY_DIR}/providers")
target_compile_definitions(Stage1Acceptance PRIVATE TEST_MATRIX_DLL=L"${CMAKE_SOURCE_DIR}/reference-runtime/EntropyInputMatrix.dll")
add_executable(EntropyMusicSourceCore Engine/App/main.cpp)
add_library(Stage1Runtime STATIC Engine/Core/Configuration.cpp Engine/Core/EngineLifecycle.cpp Engine/IPC/Stage1ControlServer.cpp Engine/IPC/Telemetry.cpp)
target_include_directories(Stage1Runtime PUBLIC Engine/Core Engine/IPC)
target_link_libraries(Stage1Runtime PUBLIC Stage1Publisher Stage1Ingress avrt)
target_link_libraries(EntropyMusicSourceCore PRIVATE Stage1Runtime)
target_compile_definitions(Stage1Acceptance PRIVATE TEST_CORE_EXE=L"${CMAKE_BINARY_DIR}/EntropyMusicSourceCore.exe" TEST_PROVIDER_DIR_W=L"${CMAKE_BINARY_DIR}/test-providers" TEST_EVIDENCE_DIR="${CMAKE_SOURCE_DIR}/evidence")
target_compile_definitions(Stage1Acceptance PRIVATE TEST_STAGE2_EXE=L"${CMAKE_SOURCE_DIR}/../Stage2/release/stage2-music-v0009/EntropyDspEngine.exe")
foreach(role music aux)
 add_library(Synthetic${role} SHARED tests/SyntheticProvider.cpp)
 target_include_directories(Synthetic${role} PRIVATE ABI)
 target_compile_definitions(Synthetic${role} PRIVATE SYNTHETIC_ID="synthetic.${role}")
 set_target_properties(Synthetic${role} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-providers")
endforeach()
add_library(BadAbiFixture SHARED tests/SyntheticProvider.cpp)
target_include_directories(BadAbiFixture PRIVATE ABI)
target_compile_definitions(BadAbiFixture PRIVATE SYNTHETIC_BAD_ABI=1 SYNTHETIC_ID="synthetic.bad-abi")
set_target_properties(BadAbiFixture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/negative-providers")
target_compile_definitions(Stage1Acceptance PRIVATE TEST_BAD_PROVIDER_DIR="${CMAKE_BINARY_DIR}/negative-providers")
foreach(adapter transport tap aux)
 add_library(Legacy${adapter} SHARED Providers/Legacy/LegacyProvider.cpp)
 target_include_directories(Legacy${adapter} PRIVATE ABI ThirdParty/NativeSrc/Shared/ModuleABI/include)
 set_target_properties(Legacy${adapter} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/providers")
endforeach()
target_compile_definitions(Legacytransport PRIVATE LEGACY_FILE=L"EntropyMusicTransportEndpoint.dll" LEGACY_ID="legacy.music-transport" LEGACY_NAME="Music Transport V1 reference" LEGACY_ROLES=1 LEGACY_ACCESS=8)
target_compile_definitions(Legacytap PRIVATE LEGACY_FILE=L"EntropyTidalEndpoint.dll" LEGACY_ID="legacy.private-tap" LEGACY_NAME="Private Tap V1 reference" LEGACY_ROLES=1 LEGACY_ACCESS=4)
target_compile_definitions(Legacyaux PRIVATE LEGACY_FILE=L"EntropyAuxEndpoint.dll" LEGACY_ID="legacy.aux" LEGACY_NAME="AUX V1 reference" LEGACY_ROLES=2 LEGACY_ACCESS=4)
add_library(WindowsCapture SHARED Providers/WindowsCapture/WindowsCaptureProvider.cpp)
target_link_libraries(WindowsCapture PRIVATE Stage1Ingress Stage1Pcm)
set_target_properties(WindowsCapture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/providers")
file(COPY reference-runtime/ DESTINATION "${CMAKE_BINARY_DIR}/providers/reference" PATTERN EntropyInputMatrix.dll EXCLUDE)
enable_testing()
add_test(NAME Stage1Pcm COMMAND Stage1Acceptance --pcm)
add_test(NAME Stage1Architecture COMMAND Stage1Acceptance --architecture)
add_test(NAME Stage1Pipeline COMMAND Stage1Acceptance --pipeline)
